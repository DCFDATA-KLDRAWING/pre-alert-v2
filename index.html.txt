        function listenToTemplates() {
            if (!currentUser || !isCloudMode) return;
            if (unsubscribeSnapshot) unsubscribeSnapshot();

                snapshot.forEach(doc => loadedTemplates.push(doc.data()));

                    state.templates = loadedTemplates.sort((a, b) => (a.createdAt || 0) - (b.createdAt || 0));

                }

            });

        function extractVariables(subject, body) {
            const combinedText = subject + "\n" + body;
            const vars = new Set();
            let match;
            while ((match = regexVars.exec(combinedText)) !== null) {
                vars.add(match[1].trim());
            }
            return Array.from(vars);
        }

        function buildForm() {
            const template = state.templates.find(t => t.id === state.currentTemplateId);
            const formContainer = document.getElementById('dynamicForm');
            formContainer.innerHTML = '';
            if (!template) return;

            state.extractedVars = extractVariables(template.subject, template.body);

            if (state.extractedVars.length === 0) {
                formContainer.innerHTML = '<div class="text-center text-gray-500 py-4">此模版沒有設定任何變數。</div>';
                return;
            }

            const gridDiv = document.createElement('div');
            gridDiv.className = 'grid grid-cols-1 md:grid-cols-2 gap-4';

            state.extractedVars.forEach(varName => {
                const wrapper = document.createElement('div');
                const label = document.createElement('label');
                label.className = 'block text-sm font-semibold text-gray-700 mb-1 truncate';
                label.textContent = varName;
                label.title = varName;

                const input = document.createElement('input');
                input.type = 'text';
                input.className = 'w-full border border-gray-300 rounded-lg p-2.5 shadow-sm focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition text-sm';
                input.placeholder = `輸入 ${varName}...`;
                input.value = state.formData[varName] || '';

                input.addEventListener('input', (e) => {
                    state.formData[varName] = e.target.value;
                    updatePreview();
                });

                wrapper.appendChild(label);
                wrapper.appendChild(input);
                gridDiv.appendChild(wrapper);
            });
            formContainer.appendChild(gridDiv);
        }

        // ★ 修改：渲染預覽畫面，如果是 HTML 就不跳脫，保留表格
        function generatePreviewHtml(text) {
            if(!text) return '';
            let html = text.replace(regexVars, (match, p1) => {
                const varName = p1.trim();
                const val = state.formData[varName];
                if (val && val.trim() !== '') {
                    return `<span class="bg-yellow-300 text-black px-1.5 py-0.5 mx-0.5 rounded shadow-sm font-bold">${escapeHtml(val)}</span>`;
                } else {
                    return `<span class="bg-gray-200 text-gray-500 px-1 mx-0.5 rounded font-mono text-xs border border-gray-300">${escapeHtml(match)}</span>`;
                }
            });
            // 判斷是不是含有 HTML (表格)
            if (!/<[a-z][\s\S]*>/i.test(html)) {
                html = html.replace(/\n/g, '<br>');
            }
            return html;
        }

        function generatePlainText(text) {
            if(!text) return '';
            return text.replace(regexVars, (match, p1) => {
                const varName = p1.trim();
                const val = state.formData[varName];
                return (val && val.trim() !== '') ? val : '';
            });
        }

        // ★ 新增：產生乾淨的 HTML (不會有黃色底色)，用來複製到信件
        function generateCleanHtml(text) {
            if(!text) return '';
            let html = text.replace(regexVars, (match, p1) => {
                const varName = p1.trim();
                const val = state.formData[varName];
                return (val && val.trim() !== '') ? escapeHtml(val) : '';
            });
            if (!/<[a-z][\s\S]*>/i.test(html)) {
                html = html.replace(/\n/g, '<br>');
            }
            return html;
        }

        // ★ 核心變更：支援把網頁表格(富文本)完美複製進剪貼簿
        function copyRichText(htmlContent, successMessage) {
            const tempDiv = document.createElement("div");
            tempDiv.innerHTML = htmlContent;
            tempDiv.style.position = "absolute";
            tempDiv.style.left = "-9999px"; // 隱藏起來
            document.body.appendChild(tempDiv);
            
            const selection = window.getSelection();
            const range = document.createRange();
            range.selectNodeContents(tempDiv);
            selection.removeAllRanges();
            selection.addRange(range);
            
            try { 
                document.execCommand('copy'); 
                showToast(successMessage); 
            } catch (err) { 
                showToast('複製失敗'); 
            }
            
            selection.removeAllRanges();
            document.body.removeChild(tempDiv);
        }

        function fallbackCopyText(text, successMessage) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            document.body.appendChild(textArea);
            textArea.select();
            try { document.execCommand('copy'); showToast(successMessage); } 
            catch (err) { showToast('複製失敗'); }
            document.body.removeChild(textArea);
        }

        async function saveTemplate() {
            const id = document.getElementById('editTemplateId').value;
            const name = document.getElementById('editTemplateName').value.trim() || '未命名模版';
            const subject = document.getElementById('editTemplateSubject').value;
            const body = document.getElementById('editTemplateBody').value;
            const timestamp = Date.now();
            let targetTemplate;

            if (id === 'new') {
                targetTemplate = { id: generateId(), name, subject, body, createdAt: timestamp };
                document.getElementById('editTemplateId').value = targetTemplate.id;
            } else {
                const existing = state.templates.find(t => t.id === id);

            if (isCloudMode && currentUser) {
                await setDoc(doc(getCollectionRef(), targetTemplate.id), targetTemplate);

            } else {
                if (id === 'new') state.templates.push(targetTemplate);

                saveToStorageLocal(); updateUI(); showToast('已儲存至本機');
            }

        function importTemplates(event) {
            const file = event.target.files[0]; if (!file) return;
            const reader = new FileReader();
            reader.onload = async function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    if (Array.isArray(data) && data.length > 0 && data[0].hasOwnProperty('id')) {
                        if (isCloudMode && currentUser) {
                            showToast('正在上傳至雲端...');
                            for (const t of data) { if (!t.createdAt) t.createdAt = Date.now(); await setDoc(doc(getCollectionRef(), t.id), t); }
                            showToast('匯入雲端成功！');
                        } else {
                            state.templates = data; saveToStorageLocal(); updateUI(); showEmptyEditor(); showToast('匯入本機成功！');
                        }
                    } else alert('檔案格式錯誤');
                } catch (error) { alert('解析失敗'); }
                event.target.value = '';
            };
            reader.readAsText(file);
        }
